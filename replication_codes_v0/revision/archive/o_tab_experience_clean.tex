\hypertarget{time-invariant-status}{%
\section{Time Invariant Status}\label{time-invariant-status}}

\begin{itemize}
\item
  Let \(n^{task}_{i}=\) Number of projects of the same task held by the
  contractor in the three quarters before starting project \(i\)
\item
  Let \(n_{i}=\) Number of projects held by the contractor in the three
  quarters before starting project \(i\)
\item
  Define:

  \begin{itemize}
  \tightlist
  \item
    Inexperienced contractor\(_i\) = 1 if \(n^{task}_{i}=0\)
  \item
    New entrant\(_i\) = 1 if \(n_{i}=0\)
  \item
    Level of inexperience\(_{i}= 1 - \frac{n^{task}_{i}}{n_{i}}\) if
    \(n_{i}>0\)
  \item
    Level of inexperience\(_{i}= 1\) if \(n_{i,k,t}=0\).
  \end{itemize}
\end{itemize}
