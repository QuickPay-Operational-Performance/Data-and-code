\begin{itemize}
\item
  Let \(n^{task}_{i}=\) Number of projects of the same task held by the
  contractor in the three quarters before starting project \(i\)
\item
  Let \(n_{i}=\) Number of projects held by the contractor in the three
  quarters before starting project \(i\)
\item
  Define:

  \begin{itemize}
  \tightlist
  \item
    Level of inexperience\(_{i}= 1 - \frac{n^{task}_{i}}{n_{i}}\) if
    \(n_{i}>0\)
  \item
    Level of inexperience\(_{i}= 1\) if \(n_{i}=0\)
  \end{itemize}
\end{itemize}

\hypertarget{discretizing}{%
\section{Discretizing}\label{discretizing}}

\begin{itemize}
\tightlist
\item
  Inexperienced \(= 1\)(Level of inexperience\(_{i}>0.5\))

  \begin{itemize}
  \tightlist
  \item
    This means less than 50\% of contractor projects in the last three
    quarters were of same task
  \end{itemize}
\item
  Inexperienced \(= 1\)(Level of inexperience\(_{i}>0.7\))

  \begin{itemize}
  \tightlist
  \item
    This means less than 30\% of contractor projects in the last three
    quarters were of same task type
  \end{itemize}
\end{itemize}
